% The document skeleton for an assembled book, in order. A template, not
% a file that compiles: copy it into content/rendered/<book>/book.tex,
% replace each <...> placeholder, keep the \input{preamble} and
% \input{copyright} lines only if the book has those files under
% content/specs/<book>/, and put each outline entry where its comment says. Each "latex-conventions.md"
% comment names the section of
% .claude/skills/book-assembler/references/latex-conventions.md that
% explains the line.
\documentclass[11pt,a4paper]{book}
\usepackage{amsmath}                         % before unicode-math and cleveref
\usepackage{fontspec}\usepackage{unicode-math}
\input{chitragupta-lualatex}                 % latex-conventions.md, "Unicode characters"
\usepackage[a4paper,margin=80pt]{geometry}   % latex-conventions.md, "Margins"
\usepackage{longtable,booktabs,array,calc}   % what the converted units use
\setlength{\LTcapwidth}{\textwidth}          % latex-conventions.md, "Table captions"
\usepackage{graphicx}
\usepackage{tikz}
\usetikzlibrary{positioning,fit}             % latex-conventions.md, "TikZ libraries"
\input{chitragupta-floats}                    % latex-conventions.md, "Figure placement"
\usepackage[hidelinks]{hyperref}
\usepackage{cleveref}
\usepackage{fvextra}                         % latex-conventions.md, "Wide code lines"
\DefineVerbatimEnvironment{verbatim}{Verbatim}{breaklines}
\usepackage[numbers,sort&compress]{natbib}   % latex-conventions.md, "The bibliography"
\setcounter{secnumdepth}{2}                  % latex-conventions.md, "Numbering"
\setcounter{tocdepth}{1}                     % chapters and sections only
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
\newcommand{\BookTitle}{<the outline's own title>}
\newcommand{\BookAuthor}{<the outline's - author: line; else ask the user, never invent one>}
\newcommand{\BuildLine}{Built <YYYY-MM-DD> \textperiodcentered{} outline <digest> \textperiodcentered{} chitragupta <version>}
\title{\BookTitle}                            % latex-conventions.md, "The copyright page"
\author{\BookAuthor}
\date{}
\input{preamble}                             % only if the book has one

\begin{document}
\frontmatter
\maketitle
\input{copyright}                            % only if the book has one
\tableofcontents
% {#front}'s entries, in outline order        % latex-conventions.md, "Front matter, appendices and back matter"

\mainmatter
% \part / \input, in outline order

\appendix\crefalias{chapter}{appendix}       % only if the outline has {#appendices}
% {#appendices}'s \input, in outline order

\backmatter
% {#back}'s entries, in outline order: units, \input{glossary}, \input{notation},
% the lists -- each generated page only if it has something to show
\bibliographystyle{IEEEtran}
\addcontentsline{toc}{chapter}{Bibliography}
\bibliography{bibliography}                  % the .bib render wrote here
\end{document}
